    <script>
        let selectedPair = "EUR/USD OTC";
        let selectedTime = "3s";

        function selectPair(el, pair) {
            document.querySelectorAll('.grid')[0].querySelectorAll('.btn-option').forEach(b => b.classList.remove('active'));
            el.classList.add('active');
            selectedPair = pair;
        }

        function selectTime(el, time) {
            document.querySelectorAll('.grid')[1].querySelectorAll('.btn-option').forEach(b => b.classList.remove('active'));
            el.classList.add('active');
            selectedTime = time;
        }

        function requestSignal() {
            const tg = window.Telegram.WebApp;
            const data = {
                pair: selectedPair,
                time: selectedTime
            };
            // Στέλνει τα επιλεγμένα δεδομένα στο Telegram Bot
            tg.sendData(JSON.stringify(data));
        }
    </script>
